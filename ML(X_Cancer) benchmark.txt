## SVM
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split, GridSearchCV, KFold
from sklearn.svm import SVC
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.preprocessing import label_binarize

data = pd.read_csv('X_Cancer.csv')
## Split data into training and testing sets
X = data.drop("Group", axis=1)  # Features
y = data["Group"] 
# Parameters
n_splits = 10
test_size = 0.3
kf = KFold(n_splits=10)  # 10-fold cross-validation for GridSearchCV
param_gridsv = {
    'C': [0.1, 1, 10, 100],
    'gamma': [0.001, 0.01, 0.1, 1],
    'kernel': ['rbf']
} 
# Initialize lists to store results
classification_reports_trainsv = []
classification_reports_testsv = []
accuracies_trainsv = []
accuracies_testsv = []
auc_scores_trainsv = []
auc_scores_testsv = []

for i in range(n_splits):
    # Split data for current iteration
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=i)
    
    # Initialize model and perform grid search
    modelsv = SVC(probability=True)
    grid_searchsv = GridSearchCV(modelsv, param_gridsv, cv=kf, scoring='accuracy')
    grid_searchsv.fit(X_train, y_train)
    best_modelsv = grid_searchsv.best_estimator_

    # Predictions and probabilities
    y_train_predsv = best_modelsv.predict(X_train)
    y_train_pred_probasv = best_modelsv.predict_proba(X_train)
    y_test_predsv = best_modelsv.predict(X_test)
    y_test_pred_probasv = best_modelsv.predict_proba(X_test)

    # Binarize labels for multi-class AUC calculation
    y_train_binsv = label_binarize(y_train, classes=np.unique(y))
    y_test_binsv = label_binarize(y_test, classes=np.unique(y))

    # Calculate AUC scores for each class and average them
    if y_train_binsv.shape[1] > 1:  # multi-class case
        auc_trainsv = roc_auc_score(y_train_binsv, y_train_pred_probasv, average="macro", multi_class="ovr")
        auc_testsv = roc_auc_score(y_test_binsv, y_test_pred_probasv, average="macro", multi_class="ovr")
    else:  # binary case
        auc_trainsv = roc_auc_score(y_train, y_train_pred_probasv[:, 1])  # for binary, select prob of class 1
        auc_testsv = roc_auc_score(y_test, y_test_pred_probasv[:, 1])

    # Append AUC scores to lists (separate row for each iteration)
    auc_scores_trainsv.append(auc_trainsv)
    auc_scores_testsv.append(auc_testsv)

    # Classification reports and accuracy
    report_trainsv = classification_report(y_train, y_train_predsv, output_dict=True)
    report_testsv = classification_report(y_test, y_test_predsv, output_dict=True)
    
    classification_reports_trainsv.append(report_trainsv)
    classification_reports_testsv.append(report_testsv)
    
    # Store accuracy
    accuracies_trainsv.append(report_trainsv["accuracy"])
    accuracies_testsv.append(report_testsv["accuracy"])
# Calculate averages for SVM model
avg_auc_trainsv = np.mean(auc_scores_trainsv)
avg_auc_testsv = np.mean(auc_scores_testsv)
avg_accuracy_trainsv = np.mean(accuracies_trainsv)
avg_accuracy_testsv = np.mean(accuracies_testsv)

# Calculate average precision, recall, and F1-score for each class in the classification reports for SVM
def avg_classification_report_svm(reports):
    avg_report = {}
    for key in reports[0].keys():
        if isinstance(reports[0][key], dict):  # For class-specific metrics
            avg_report[key] = {}
            for metric in reports[0][key].keys():
                avg_report[key][metric] = np.mean([r[key][metric] for r in reports])
        else:
            avg_report[key] = np.mean([r[key] for r in reports])  # For overall metrics
    return avg_report

avg_classification_report_trainsv = avg_classification_report_svm(classification_reports_trainsv)
avg_classification_report_testsv = avg_classification_report_svm(classification_reports_testsv)


# Convert average SVM test classification report to DataFrame
avg_classification_report_testsv_df = pd.DataFrame(avg_classification_report_testsv).T

# Display the DataFrame
print("Average SVM Test Classification Report:")
print(avg_classification_report_testsv_df)


# Convert average SVM test classification report to DataFrame
avg_classification_report_trainsv_df = pd.DataFrame(avg_classification_report_trainsv).T

# Display the DataFrame
print("Average SVM Train Classification Report:")
print(avg_classification_report_trainsv_df)

#Random Undersampling

import pandas as pd
from sklearn.utils import resample
df=pd.read_csv('Breast_Cancer.csv')

# Separate majority and minority classes
df_majority = df[df['Group'] == df['Group'].value_counts().idxmax()]
df_minority = df[df['Group'] != df['Group'].value_counts().idxmax()]

# Downsample majority class
df_majority_downsampled = resample(df_majority, 
                                   replace=False,    # sample without replacement
                                   n_samples=int(len(df_minority) * 0.65),  # set imbalance level
                                   random_state=42)  

# Combine minority class with downsampled majority class
df_imbalanced = pd.concat([df_majority_downsampled, df_minority])

# Shuffle the data
df_imbalanced = df_imbalanced.sample(frac=1, random_state=42).reset_index(drop=True)

# Check new class distribution
print("Class distribution after imbalance:\n", df_imbalanced['Group'].value_counts())
